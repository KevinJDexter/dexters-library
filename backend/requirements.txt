# Python dependencies for the Dexter's Library API.
#
# Install with:  pip install -r requirements.txt
#
# Versions are left open for now. Once we've confirmed a working install,
# we'll pin exact versions here so the deployed app matches your machine.

fastapi          # The web framework — turns Python functions into HTTP endpoints
uvicorn[standard]  # The web server that actually runs FastAPI
sqlmodel         # ORM — lets us talk to Postgres as Python objects instead of raw SQL
psycopg[binary]  # PostgreSQL driver. [binary] skips compiling C code on install
python-dotenv    # Loads backend/.env into the environment for local dev
pytest           # Test runner — finds test_*.py files and runs their functions
httpx            # HTTP client; FastAPI's TestClient needs it to make in-process requests
python-multipart # Parses multipart/form-data — required for file uploads (CSV import)
